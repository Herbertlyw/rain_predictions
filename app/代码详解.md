#### 4.1 ConvLSTMCell（卷积LSTM单元）

```101:132:app.py
class ConvLSTMCell(nn.Module):
    def __init__(self, input_dim, hidden_dim, kernel_size, bias=True):
        super(ConvLSTMCell, self).__init__()
        self.input_dim = input_dim
        self.hidden_dim = hidden_dim
        self.kernel_size = kernel_size
        self.padding = kernel_size // 2
        self.bias = bias

        self.conv = nn.Conv2d(in_channels=input_dim + hidden_dim,
                              out_channels=4 * hidden_dim,
                              kernel_size=kernel_size,
                              padding=self.padding,
                              bias=bias)

    def forward(self, input_tensor, cur_state):
        h_cur, c_cur = cur_state
        
        combined = torch.cat([input_tensor, h_cur], dim=1)
        combined_conv = self.conv(combined)
        
        cc_i, cc_f, cc_o, cc_g = torch.split(combined_conv, self.hidden_dim, dim=1)
        i = torch.sigmoid(cc_i)
        f = torch.sigmoid(cc_f)
        o = torch.sigmoid(cc_o)
        g = torch.tanh(cc_g)

        c_next = f * c_cur + i * g
        h_next = o * torch.tanh(c_next)
        
        return h_next, c_next
```

**功能说明：**
- **ConvLSTM** 是结合了卷积和 LSTM 的单元，用于处理空间-时间序列数据
- 输入和隐藏状态通过卷积操作融合
- 输出4个门控信号：输入门(i)、遗忘门(f)、输出门(o)、候选值(g)
- 更新细胞状态(c)和隐藏状态(h)

```134:199:app.py
class PredRNN(nn.Module):
    def __init__(self, config):
        super(PredRNN, self).__init__()
        self.config = config
        num_hidden = [int(x) for x in config.num_hidden.split(',')]
        self.num_layers = len(num_hidden)
        self.num_hidden = num_hidden
        
        cell_list = []
        for i in range(self.num_layers):
            in_channel = 1 if i == 0 else num_hidden[i-1]
            cell_list.append(ConvLSTMCell(in_channel, num_hidden[i], config.filter_size))
        self.cell_list = nn.ModuleList(cell_list)
        
        self.conv_last = nn.Conv2d(num_hidden[-1], 1, kernel_size=1, stride=1, padding=0, bias=False)
        
        # 初始化权重
        self._initialize_weights()

    def _initialize_weights(self):
        for m in self.modules():
            if isinstance(m, nn.Conv2d):
                nn.init.kaiming_normal_(m.weight, mode='fan_out', nonlinearity='relu')
                if m.bias is not None:
                    nn.init.constant_(m.bias, 0)

    def forward(self, frames_tensor, mask_true=None, return_loss=True):
        device = frames_tensor.device
        frames = frames_tensor.unsqueeze(2)
        
        batch_size, seq_len, _, height, width = frames.shape
        next_frames = []
        
        # 初始化隐藏状态
        h = [torch.zeros(batch_size, self.num_hidden[i], height, width, device=device) 
             for i in range(self.num_layers)]
        c = [torch.zeros(batch_size, self.num_hidden[i], height, width, device=device) 
             for i in range(self.num_layers)]
        
        for t in range(seq_len - 1):
            net = frames[:, t]
            
            # 第一层
            h[0], c[0] = self.cell_list[0](net, (h[0], c[0]))
            
            # 后续层
            for i in range(1, self.num_layers):
                h[i], c[i] = self.cell_list[i](h[i-1], (h[i], c[i]))
            
            # 生成预测
            x_gen = self.conv_last(h[-1])
            next_frames.append(x_gen)
        
        next_frames = torch.stack(next_frames, dim=1)
        
        if return_loss:
            target = frames[:, 1:]
            if torch.all(target == 0):
                logger.warning("目标帧全为0，可能数据有问题")
                return next_frames.squeeze(2), torch.tensor(0.0, device=device)
            
            loss = F.mse_loss(next_frames, target)
            return next_frames.squeeze(2), loss
        
        return next_frames.squeeze(2), None
```

**模型结构：**
- **多层 ConvLSTM**: 堆叠多个 ConvLSTM 单元，逐层处理特征
- **时序处理**: 逐时间步处理输入序列，每个时间步生成一个预测帧
- **输出层**: 使用 1x1 卷积将隐藏特征映射回原始图像空间
- **损失计算**: 使用 MSE（均方误差）损失函数

```205:241:app.py
        model.eval()
```

```32:58:train.py
def apply_stability_measures(tensor, name=""):
    """应用全面的数值稳定性措施"""
    if tensor is None:
        return tensor
        
    if torch.isnan(tensor).any() or torch.isinf(tensor).any():
        logger.warning(f"NaN/Inf detected in {name}, applying stability measures")
        
        # 1. 尝试用均值替换
        tensor_mean = torch.nanmean(tensor)
        if torch.isnan(tensor_mean) or torch.isinf(tensor_mean):
            tensor_mean = torch.tensor(0.0, device=tensor.device)
        
        tensor = torch.where(torch.isfinite(tensor), tensor, tensor_mean)
        
        # 2. 如果仍有问题，使用0替换
        tensor = torch.nan_to_num(tensor, nan=0.0, posinf=1.0, neginf=0.0)
    
    # 3. 添加微小噪声防止梯度消失
    noise = torch.randn_like(tensor) * 1e-6
    tensor = tensor + noise
    
    # 4. 限制数值范围
    tensor = torch.clamp(tensor, -50.0, 50.0)
    
    return tensor
```

**功能说明：**
- **NaN/Inf 检测**: 检测并处理无效数值
- **均值填充**: 用有效值的均值替换无效值
- **噪声注入**: 添加微小噪声防止梯度消失
- **范围限制**: 将数值限制在合理范围内（-50 到 50）

```60:156:train.py
class SpatioTemporalLSTMCellv2(nn.Module):
    def __init__(self, in_channel, num_hidden, height, width, filter_size, stride, layer_norm):
        super(SpatioTemporalLSTMCellv2, self).__init__()

        self.num_hidden = num_hidden
        self.padding = filter_size // 2
        self._forget_bias = 1.0
        
        # 使用深度可分离卷积减少参数数量
        def create_conv_block(in_c, out_c):
            layers = []
            layers.append(
                nn.Conv2d(in_c, out_c, kernel_size=filter_size,
                          stride=stride, padding=self.padding, bias=False)
            )
            if layer_norm:
                layers.append(nn.LayerNorm([out_c, height, width]))
            else:
                layers.append(nn.BatchNorm2d(out_c))
            layers.append(nn.ReLU(inplace=True))
            return nn.Sequential(*layers)
        
        # 使用更高效的卷积块
        self.conv_x = create_conv_block(in_channel, num_hidden * 7)
        self.conv_h = create_conv_block(num_hidden, num_hidden * 4)
        self.conv_m = create_conv_block(num_hidden, num_hidden * 3)
        self.conv_o = create_conv_block(num_hidden * 2, num_hidden)
        
        self.conv_last = nn.Sequential(
            nn.Conv2d(num_hidden * 2, num_hidden, kernel_size=1, stride=1, padding=0, bias=False),
            nn.BatchNorm2d(num_hidden),
            nn.ReLU(inplace=True)
        )
        
        # 初始化权重
```

```108:156:train.py
        splits = [self.num_hidden] * 4
        i_h, f_h, g_h, o_h = torch.split(h_concat, splits, dim=1)
        
        splits = [self.num_hidden] * 3
        i_m, f_m, g_m = torch.split(m_concat, splits, dim=1)

        i_t = torch.sigmoid(i_x + i_h)
        f_t = torch.sigmoid(f_x + f_h + self._forget_bias)
        g_t = torch.tanh(g_x + g_h)

        delta_c = i_t * g_t
        delta_c = apply_stability_measures(delta_c, "delta_c")
        c_new = f_t * c_t + delta_c
        c_new = apply_stability_measures(c_new, "c_new")

        i_t_prime = torch.sigmoid(i_x_prime + i_m)
        f_t_prime = torch.sigmoid(f_x_prime + f_m + self._forget_bias)
        g_t_prime = torch.tanh(g_x_prime + g_m)

        delta_m = i_t_prime * g_t_prime
        delta_m = apply_stability_measures(delta_m, "delta_m")
        m_new = f_t_prime * m_t + delta_m
        m_new = apply_stability_measures(m_new, "m_new")

        mem = torch.cat((c_new, m_new), 1)
        o_t = torch.sigmoid(o_x + o_h + self.conv_o(mem))
        h_new = o_t * torch.tanh(self.conv_last(mem))
        
        # 最终稳定性检查
        h_new = apply_stability_measures(h_new, "h_new")
```

1. **输入处理**: 对输入、隐藏状态、细胞状态、记忆状态应用稳定性措施
2. **卷积特征提取**: 分别对输入、隐藏状态、记忆状态进行卷积
3. **门控计算**: 计算输入门、遗忘门、候选值等
4. **状态更新**: 更新细胞状态(c)和记忆状态(m)
5. **输出生成**: 结合更新后的状态生成新的隐藏状态

```217:233:train.py
    def _ssim_loss(self, x, y):
        """计算SSIM损失"""
        C1 = 0.01 ** 2
        C2 = 0.03 ** 2
        
        mu_x = F.avg_pool2d(x, 3, 1, padding=1)
        mu_y = F.avg_pool2d(y, 3, 1, padding=1)
        
        sigma_x = F.avg_pool2d(x**2, 3, 1, padding=1) - mu_x**2
        sigma_y = F.avg_pool2d(y**2, 3, 1, padding=1) - mu_y**2
        sigma_xy = F.avg_pool2d(x*y, 3, 1, padding=1) - mu_x*mu_y
        
        ssim_n = (2*mu_x*mu_y + C1) * (2*sigma_xy + C2)
        ssim_d = (mu_x**2 + mu_y**2 + C1) * (sigma_x + sigma_y + C2)
        
        ssim = ssim_n / ssim_d
        return torch.clamp((1 - ssim) / 2, 0, 1).mean()
```

**SSIM说明：**
- **结构相似性指数**: 衡量图像的结构相似性，比MSE更能反映人眼感知
- **计算方式**: 比较局部窗口的均值、方差和协方差
- **损失值**: 1 - SSIM，范围 [0, 1]

5. **保存结果**: 保存为 `.npy` 文件和可视化图像

## 总结

### app.py 和 train.py 的关系

1. **train.py**: 负责模型训练，生成 `best_model.pth` 或 `predrnn_v2_best.pth`
2. **app.py**: 使用训练好的模型提供 Web 服务，接收用户上传的数据并进行预测

### 主要区别

| 特性 | app.py | train.py |
|------|--------|----------|
| **模型架构** | 简化的 PredRNN（ConvLSTM） | 完整的 PredRNNv2（SpatioTemporalLSTM） |
| **用途** | Web服务、实时预测 | 模型训练、离线预测 |
| **数据处理** | 简单的归一化 | 全局归一化、数据增强、Patch处理 |
| **损失函数** | MSE | MSE + SSIM + Decouple Loss |
| **训练特性** | 无 | Schedule Sampling、梯度累积、混合精度 |

### 关键技术点

1. **时空序列预测**: 使用 LSTM 处理时间序列，卷积处理空间特征
2. **数值稳定性**: 全面的 NaN/Inf 检测和处理机制
3. **混合精度训练**: 使用 AMP 加速训练并节省显存
4. **Schedule Sampling**: 逐步从使用真实值过渡到使用预测值，提高泛化能力
5. **多损失函数**: 结合 MSE、SSIM 和去耦损失，提高预测质量

### 使用建议

1. **训练模型**: 使用 `train.py` 训练模型，调整超参数
2. **部署服务**: 使用 `app.py` 部署 Web 服务，提供在线预测
3. **模型兼容**: 注意两个文件使用的模型架构不同，需要确保模型文件匹配
